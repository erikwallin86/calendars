# Lägg t.ex. i B10, tar information från Matcher fliken, och gör så att det blir en match per rad.
=MAP(
  Matcher!F8:F;
  Matcher!G8:G;
  Matcher!H8:H;
  LAMBDA(hemma; borta; matchtid;
    IF(
      ISDATE(matchtid);
      TEXTJOIN(CHAR(10); TRUE;
        "BEGIN:VEVENT";
        "UID:" & TEXT(matchtid;"yyyymmdd\THHmmss") & "-" & ROW(matchtid) & "@guif";
        "DTSTAMP:" & TEXT(NOW();"yyyymmdd\THHmmss") & "Z";

        "SUMMARY:Match " & hemma & " - " & borta &
          " " & TEXT(matchtid;"HH:mm") &
          ", samling " & TEXT(matchtid-TIME(0;30;0);"HH:mm");

        "DTSTART;TZID=Europe/Stockholm:" &
          TEXT(matchtid-TIME(0;30;0);"yyyymmdd\THHmmss");

        "DTEND;TZID=Europe/Stockholm:" &
          TEXT(matchtid+TIME(1;0;0);"yyyymmdd\THHmmss");

        "END:VEVENT"
      );
      ""
    )
  )
)
